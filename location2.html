            userLocation.lon = position.coords.longitude;

    // Load nearby facilities using Overpass API

    function loadNearbyFacilities(lat, lon) {

      const facilitiesList = document.getElementById('facilities-list');

      facilitiesList.innerHTML = '<div class="loading">Searching for facilities...</div>';

      

      // Clear existing facility markers

      facilitiesMarkers.forEach(marker => marker.remove());

      facilitiesMarkers = [];

      

      const facilities = [];

      const radius = 1000; // 1km in meters

      

      // Define search types with Overpass queries

      const queries = [

        { type: 'Public Toilet', query: 'node[amenity=toilets]', icon: '🚻' },

        { type: 'Bus Station', query: 'node[amenity=bus_station]', icon: '🚌' },

        { type: 'Train Station', query: 'node[amenity=train_station]', icon: '🚂' },

        { type: 'Supermarket', query: 'node[shop=supermarket]', icon: '🛒' },

        { type: 'Restaurant', query: 'node[amenity=restaurant]', icon: '🍽️' }

      ];

      

      let completedQueries = 0;

      

      queries.forEach(item => {

        const bbox = `${lon - 0.009},${lat - 0.009},${lon + 0.009},${lat + 0.009}`; // ~1km bbox

        const overpassUrl = `https://overpass-api.de/api/interpreter?data=[bbox:${bbox}];${item.query};out center;`;

        

        fetch(overpassUrl)

          .then(response => response.json())

          .then(data => {

            if (data.elements) {

              data.elements.forEach(element => {

                const elemLat = element.center?.lat || element.lat;

                const elemLon = element.center?.lon || element.lon;

                

                if (elemLat && elemLon) {

                  const distance = calculateDistance(lat, lon, elemLat, elemLon);

                  

                  if (distance <= 1) { // 1 km

                    facilities.push({

                      type: item.type,

                      icon: item.icon,

                      name: element.tags?.name || `${item.type} #${facilities.length + 1}`,

                      lat: elemLat,

                      lon: elemLon,

                      distance: distance

                    });

                    

                    // Add marker to map

                    const markerColor = ['red', 'green', 'orange', 'blue', 'purple'][queries.indexOf(item)];

                    const marker = L.marker([elemLat, elemLon], {

                      icon: L.icon({

                        iconUrl: `https://raw.githubusercontent.com/pointhi/leaflet-color-markers/master/img/marker-icon-2x-${markerColor}.png`,

                        shadowUrl: 'https://cdnjs.cloudflare.com/ajax/libs/leaflet/0.7.7/images/marker-shadow.png',

                        iconSize: [25, 41],

                        iconAnchor: [12, 41],

                        popupAnchor: [1, -34],

                        shadowSize: [41, 41]

                      })

                    }).addTo(map).bindPopup(`<b>${item.icon} ${element.tags?.name || item.type}</b><br>${distance.toFixed(2)} km`);

                    

                    facilitiesMarkers.push(marker);

                  }

                }

              });

            }

            

            completedQueries++;

            if (completedQueries === queries.length) {

              displayFacilities(facilities);

            }

          })

          .catch(error => {

            console.error('Overpass API error:', error);

            completedQueries++;

            if (completedQueries === queries.length) {

              displayFacilities(facilities);

            }

          });

      });

    }

    

    // Calculate distance between two coordinates (in km)

    function calculateDistance(lat1, lon1, lat2, lon2) {

      const R = 6371; // Earth's radius in km

      const dLat = (lat2 - lat1) * Math.PI / 180;

      const dLon = (lon2 - lon1) * Math.PI / 180;

      const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +

                Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *

                Math.sin(dLon / 2) * Math.sin(dLon / 2);

      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

      return R * c;

    }

    

    // Display facilities in sidebar

    function displayFacilities(facilities) {

      const facilitiesList = document.getElementById('facilities-list');

      

      if (facilities.length === 0) {

        facilitiesList.innerHTML = '<div class="no-items">No facilities found nearby</div>';

        return;

      }

      

      // Sort by distance

      facilities.sort((a, b) => a.distance - b.distance);

      

      let html = '';

      facilities.forEach(facility => {

        html += `

          <div class="facility-item">

            <div class="facility-name">${facility.icon} ${facility.name}</div>

            <span class="facility-type">${facility.type}</span>

            <div class="facility-distance">📏 ${facility.distance.toFixed(2)} km away</div>

          </div>

        `;

      });

      

      facilitiesList.innerHTML = html;

    }

    

    // Add friend

    function addFriend() {

      const name = document.getElementById('friend-name').value.trim();

      const contact = document.getElementById('friend-contact').value.trim();

    }

    // Delete friend

    function deleteFriend(id) {

      if (confirm('Delete this friend?')) {

        friends = friends.filter(f => f.id !== id);

        localStorage.setItem('friends', JSON.stringify(friends));

        

        if (markers[`friend-${id}`]) {

          markers[`friend-${id}`].remove();

        }

        

        displayFriends();

      }

    }

    // Copy to clipboard

    function copyToClipboard() {

      const shareLink = document.getElementById('share-link').textContent;

      if (!shareLink) {

        alert('Please generate a share link first');

        return;

      }

      

      navigator.clipboard.writeText(shareLink).then(() => {
